    Estimator API¶

    The pyrox_nn.api subpackage exposes sklearn-style immutable estimator facades. Each estimator wraps a pyrox_nn model + the inference runners in pyrox.inference behind a one-call fit/predict ergonomic.

    class BNFEstimator(EstimatorBase):
    r"""MAP-fit Bayesian Neural Field estimator.

    Composes the BNF layer (`pyrox_nn.BayesianNeuralField`) with
    pandas-side preprocessing (`pyrox_nn.preprocessing.fit_spatiotemporal`)
    and an ensemble-MAP inference loop
    (`pyrox.inference.ensemble_map`).

    Attributes:
        feature_cols: Input columns (first one is time).
        target_col: Target column.
        width: Hidden MLP width.
        depth: Hidden MLP depth.
        observation_model: ``"NORMAL"`` (currently only this is fully
            implemented). ``"NB"`` and ``"ZINB"`` raise
            ``NotImplementedError`` until ``gaussx.mixture_quantile``
            ships.
        seasonality_periods: Periods for seasonal features. Empty ⇒
            no seasonal features.
        num_seasonal_harmonics: Harmonics per period.
        fourier_degrees: Per-input dyadic Fourier degrees. Length must
            match ``feature_cols`` (or empty ⇒ no Fourier features).
        interactions: Pair-index list for interaction features.
        encoder: Optional single-example encoder ``(C,) -> (E,)`` whose
            output enters the network as one extra gated feature block (see
            `pyrox_nn.BayesianNeuralField`), for example
            ``geonnax.SphericalHarmonicEncoder(l_max=10, input_mode="lonlat")``.
            It receives ``encoder_cols`` in the DataFrame's own units: the
            estimator undoes its standardization of those columns first (the
            time column arrives encoded, as in the design matrix).
        encoder_cols: Indices into ``feature_cols`` fed to ``encoder``, in
            order; required exactly when ``encoder`` is given.
        timetype: ``"int"`` or ``"datetime"``.
        freq: Optional unit string for ``datetime`` time columns.
        noise: ``"fixed"`` (default) keeps the noise scale $\sigma_0$ at
            ``sigma_obs``; ``"learned"`` infers it through the sample site
            ``log_sigma_obs`` $= \log \sigma_0 \sim \mathcal N(\log
            \sigma_{\text{obs}}, 1)$, which the MAP and VI ensembles treat
            like any other latent. With ``prior_weight=0`` (MLE) and no
            ``sigma_col`` nothing bounds $\sigma_0$ below, so that
            combination raises ``ValueError``.
        sigma_obs: Observation noise std $\sigma_0$ when ``noise="fixed"``
            (with ``sigma_col``, a floor for representativeness error), or
            the prior median of $\sigma_0$ when ``noise="learned"``.
        sigma_col: Optional column of per-row noise stds $\sigma_i > 0$, in
            target units. Each row's likelihood is
            $\mathcal N(f(x_i), \sigma_i^2 + \sigma_0^2)$. The target is not
            rescaled by the estimator, so $\sigma_i$ is used as given.
        ensemble_size: Number of MAP ensemble members.
        num_epochs: SGD-MAP iterations per member.
        learning_rate: Adam learning rate.
        prior_weight: ``1`` ⇒ MAP, ``0`` ⇒ MLE, fractional ⇒ tempered.
    """

    width: int = eqx.field(static=True, default=64)
    depth: int = eqx.field(static=True, default=4)
    observation_model: ObservationModel = eqx.field(static=True, default="NORMAL")
    seasonality_periods: tuple[float, ...] = eqx.field(static=True, default=())
    num_seasonal_harmonics: tuple[int, ...] = eqx.field(static=True, default=())
    fourier_degrees: tuple[int, ...] = eqx.field(static=True, default=())
    interactions: tuple[tuple[int, int], ...] = eqx.field(static=True, default=())
    encoder: Callable[[Array], Array] | None = None
    encoder_cols: tuple[int, ...] | None = eqx.field(static=True, default=None)
    timetype: Literal["int", "datetime"] = eqx.field(static=True, default="int")
    freq: str | None = eqx.field(static=True, default=None)
    noise: NoiseModel = eqx.field(static=True, default="fixed")
    sigma_obs: float = eqx.field(static=True, default=0.1)
    sigma_col: str | None = eqx.field(static=True, default=None)
    ensemble_size: int = eqx.field(static=True, default=16)
    num_epochs: int = eqx.field(static=True, default=2000)
    learning_rate: float = eqx.field(static=True, default=5e-3)
    prior_weight: float = eqx.field(static=True, default=1.0)
    inference_kind: InferenceKind = eqx.field(static=True, default="map")
    standardize_columns: tuple[str, ...] | None = eqx.field(static=True, default=None)
    # Optional guide factory for the VI path. Takes the numpyro model
    # function and returns a guide. Defaults to AutoNormal at fit-time
    # (mean-field). Pass `AutoLowRankMultivariateNormal`, `AutoIAFNormal`,
    # `AutoBNAFNormal`, or any other AutoGuide constructor for richer
    # variational families that can represent posterior weight
    # correlations the mean-field guide cannot.
    guide_factory: Callable[[Callable], Any] | None = eqx.field(
        static=True, default=None
    )

    def fit(self, df: pd.DataFrame, *, seed: PRNGKeyArray | int) -> FittedBNF:
        if self.observation_model != "NORMAL":
            raise NotImplementedError(
                f"observation_model={self.observation_model!r} requires "
                "gaussx.mixture_quantile (tracked as gaussx#121). Use "
                "observation_model='NORMAL' for now."
            )
        if self.noise not in ("fixed", "learned"):
            # A BayesianNeuralField returns the field alone, so it cannot be
            # heteroscedastic.
            raise ValueError(f"noise must be 'fixed' or 'learned'; got {self.noise!r}.")
        # sigma_obs, the inference kind and learned noise under MLE are
        # validated by NeuralFieldEstimator.fit, after the preprocessing.
        # 1. Pandas-side preprocessing → SpatiotemporalFit bundle.
        fit_bundle = fit_spatiotemporal(
            df,
            feature_cols=self.feature_cols,
            target_col=self.target_col,
            timetype=self.timetype,
            freq=self.freq,
            seasonality_periods=self.seasonality_periods,
            num_seasonal_harmonics=self.num_seasonal_harmonics,
            fourier_degrees=self.fourier_degrees,
            interactions=self.interactions,
            standardize=self.standardize_columns,
        )
        # 2. Build the BNF layer template + design matrix.
        bnf = _build_bnf(
            fit_bundle, self.width, self.depth, self.encoder, self.encoder_cols
        )
        x_train = _df_to_design(df, fit_bundle, timetype=self.timetype, freq=self.freq)
        y_train = jnp.asarray(df[self.target_col].to_numpy(), dtype=jnp.float32)
        sigma_i = None
        if self.sigma_col is not None:
            sigma_i = _sigma_column(df, self.sigma_col)
        # 3. Run ensemble inference: a NeuralFieldEstimator on the BNF.
        fitted = NeuralFieldEstimator(
            model_fn=lambda: bnf,
            noise=self.noise,
            sigma_obs=self.sigma_obs,
            ensemble_size=self.ensemble_size,
            inference=self.inference_kind,
            num_epochs=self.num_epochs,
            learning_rate=self.learning_rate,
            prior_weight=self.prior_weight,
            guide_factory=self.guide_factory,
        ).fit(x_train, y_train, sigma=sigma_i, seed=seed)
        return FittedBNF(
            config=self,
            fit_bundle=fit_bundle,
            params=fitted.params,
            losses=fitted.losses,
            bnf_template=bnf,
        )

    def predict(
        self,
        df: pd.DataFrame,
        *,
        quantiles: Sequence[float] | None = None,
        predict_seed: int = 0,
        num_posterior_samples: int = 32,
    ) -> Float[Array, " N"] | tuple[Float[Array, " N"], tuple[Float[Array, " N"], ...]]:
        r"""Posterior-predictive mean (and optional quantiles).

        For MAP fits the per-member predictions are point estimates;
        for VI fits we draw ``num_posterior_samples`` weight samples
        per member from the variational guide.

        The quantiles are those of the latent field plus the noise scale
        $\sigma_0$ (the fixed ``sigma_obs``, or each member's or draw's
        learned value): an equal-weight mixture of
        $\mathcal N(f_m(x), \sigma_{0,m}^2)$. Per-row $\sigma_i$ from
        ``sigma_col`` are not known at new points, so they are not added;
        add $\sigma_i^2$ yourself where you know it.

        Args:
            df: Inputs.
            quantiles: Optional levels. Returns ``(mean, (q_1, ..., q_K))``.
            predict_seed: PRNG seed for VI posterior sampling and
                quantile MC.
            num_posterior_samples: Per-member posterior draws used for
                VI predictive moments and quantile MC.
        """
        cfg = self.config
        assert isinstance(cfg, BNFEstimator)
        x_pred = _df_to_design(
            df, self.fit_bundle, timetype=cfg.timetype, freq=cfg.freq
        )
        return _predict(
            self.params,
            self.bnf_template,
            x_pred,
            encoder=None,
            noise=cfg.noise,
            sigma_obs=cfg.sigma_obs,
            has_sigma=cfg.sigma_col is not None,
            inference=cfg.inference_kind,
            guide_factory=cfg.guide_factory,
            quantiles=quantiles,
            predict_seed=predict_seed,
            num_posterior_samples=num_posterior_samples,
            # The BNF estimator keeps its published 200-draw floor, so its
            # quantiles are unchanged; NeuralFieldEstimator uses 2000.
            min_quantile_samples=200,
        )

    class NeuralFieldEstimator(eqx.Module):
    r"""Fit any Bayesian neural field with ensemble MAP or VI.

    The field is built by ``model_fn()``: a callable mapping a batch of
    (encoded) coordinates ``(N, D_enc)`` to the field ``(N,)`` or
    ``(N, 1)`` while registering its weights as NumPyro sample sites,
    usually a `PyroxModule` such as `pyrox_nn.BayesianSIREN`,
    `pyrox_nn.SphericalHarmonicFeatures` or
    `pyrox_nn.BayesianNeuralField`. Each ensemble member maximises (MAP) or
    approximates (VI) the posterior of

    $$
    y_i \sim \mathcal N\bigl(f_\theta(\gamma(x_i)),\
        \sigma_i^2 + \sigma_0^2\bigr),
    $$

    where $\gamma$ is ``encoder`` (identity when ``None``), $\sigma_i$ the
    optional per-row noise passed to ``fit`` and $\sigma_0$ set by
    ``noise``. Rows with ``valid[i] = False`` contribute nothing to the log
    joint or its gradient, so padded batches fit exactly like their valid
    rows alone.

    Sites: the field's own, plus ``log_sigma_obs`` with
    ``noise="learned"``; the likelihood site ``y`` and the deterministic
    sites ``f`` (and ``log_sigma_f`` with ``noise="heteroscedastic"``).

    Attributes:
        model_fn: Zero-argument factory returning the field. Called once per
            ``fit``; the returned module's sites must not depend on the data.
        encoder: Optional deterministic single-row encoder
            $\gamma: (D,) \to (D_\mathrm{enc},)$ applied to every row before
            the field (vmapped), e.g.
            ``geonnax.SphericalHarmonicEncoder(l_max=20, input_mode="lonlat")``
            (lon/lat in radians).
        noise: ``"fixed"`` keeps $\sigma_0$ at ``sigma_obs``; ``"learned"``
            infers it through the site ``log_sigma_obs`` $= \log\sigma_0 \sim
            \mathcal N(\log\sigma_\mathrm{obs}, 1)$; ``"heteroscedastic"``
            uses the field's own input-dependent scale: the field must
            return ``(mu, log_sigma)``, e.g. a trunk followed by
            `pyrox_nn.GaussianRegressionHead`, and $\sigma_0 = \sigma(x_i)$.
            MAP members start from prior draws, and the head clips
            $\log\sigma$ to $[-7, 3]$ with zero gradient outside, so keep its
            ``log_scale_prior_std`` small (the default 0.1): with 1.0 and a
            SIREN trunk, members start past the clip and never leave it.

            nothing bounds $\sigma_0$ below, so ``fit`` raises.

            $\sigma_0$ for ``noise="learned"``; unused when heteroscedastic.

        inference: ``"map"`` (``pyrox.inference.ensemble_map``) or ``"vi"``

        num_epochs: Optimiser steps per member (full batch).
        learning_rate: Adam learning rate.
        prior_weight: ``1`` ⇒ MAP, ``0`` ⇒ MLE, fractional ⇒ tempered

            ``None`` ⇒ ``AutoNormal``.

    def fit(
        self,
        X: Float[ArrayLike, "N D"],
        y: Float[ArrayLike, " N"],
        *,
        sigma: Float[ArrayLike, " N"] | None = None,
        valid: ArrayLike | None = None,
        seed: PRNGKeyArray | int,

        r"""Fit the ensemble on arrays.

        Args:
            X: Coordinates of shape ``(N, D)`` (the encoder's input units).
            y: Targets of shape ``(N,)``.
            sigma: Optional per-row noise stds $\sigma_i$ of shape ``(N,)``,

            sigma_obs=self.sigma_obs,

            prior_weight=self.prior_weight,
            has_sigma=sigma is not None,
        )

        if isinstance(seed, int):
            seed = jr.PRNGKey(seed)
        field = self.model_fn()
        model_fn = _field_model(
            _make_forward(field, self.encoder),
            self.sigma_obs,
            noise=self.noise,
            has_sigma=sigma is not None,
            has_valid=valid is not None,
        )
        params, losses = _fit_ensemble(
            model_fn,
            x_train,
            y_train,
            inference=self.inference,
            ensemble_size=self.ensemble_size,
            num_epochs=self.num_epochs,
            learning_rate=self.learning_rate,
            prior_weight=self.prior_weight,
            guide_factory=self.guide_factory,
            seed=seed,
        )
        return FittedNeuralField(
            config=self,
            field=field,
            params=params,
            losses=losses,
            has_sigma=sigma is not None,
        )

    def fit_xarray(
        self,
        ds: Any,
        *,
        coords: Sequence[str],
        target: str,
        sigma: str | None = None,
        time_origin: Any = None,
        seed: PRNGKeyArray | int,
    ) -> FittedNeuralField:
        r"""Fit on an xarray Dataset (``pip install pyrox-nn[xarray]``).

        Every variable named in ``coords``, ``target`` and ``sigma`` (a
        dimension, coordinate or data variable) is broadcast against the
        others and flattened, in the target's dimension order, to one row
        per grid point; ``coords`` become the columns of ``X`` in the order
        given. Rows where any of them is missing (NaN, NaT) are masked out,
        so a gappy grid needs no reshaping.

        A datetime coordinate becomes float days since ``time_origin``
        (default: its earliest timestamp), so the column starts near 0
        instead of at ~2·10⁴ days since 1970, which float32 would resolve
        only to minutes and a SIREN would see as a huge frequency. A
        timedelta coordinate becomes float days. The origins are stored on
        the fit; `FittedNeuralField.predict_xarray` applies the same
        mapping, and ``predict`` on arrays expects the same units.

        Args:
            ds: An ``xarray.Dataset``.
            coords: Names of the input variables, e.g. ``("lon", "lat", "time")``.
            target: Name of the target variable.
            sigma: Optional name of the per-row noise std variable.
            time_origin: Optional reference time for every datetime
                coordinate (anything ``numpy.datetime64`` accepts).
            seed: PRNG key or integer seed.

        Returns:
            The fitted ensemble, which remembers ``coords`` and the time
            origins.

        Raises:
            ImportError: If xarray is not installed.
            ValueError: If a name is not a variable of ``ds``, or as `fit`.
        """
        xr = _require_xarray("fit_xarray")
        names = [*coords, target, *([sigma] if sigma is not None else [])]
        _check_names(ds, names)
        # Flatten in the target's dimension order (C order), whatever the
        # order of ``coords``.
        arrays = xr.broadcast(*(ds[name] for name in names))
        dims = (
            *ds[target].dims,
            *(d for d in arrays[0].dims if d not in ds[target].dims),
        )
        arrays = [a.transpose(*dims) for a in arrays]
        origins = tuple(_time_origin(ds[name].values, time_origin) for name in coords)
        cols = [
            _numeric_column(a.values.ravel(), origin)
            for a, origin in zip(
                arrays, (*origins, *([None] * (len(names) - len(coords)))), strict=True
            )
        ]
        X = np.stack(cols[: len(coords)], axis=-1)
        valid = np.isfinite(np.stack(cols, axis=-1)).all(axis=-1)
        fitted = self.fit(
            X,
            cols[len(coords)],
            sigma=None if sigma is None else cols[-1],
            valid=valid,
            seed=seed,
        )
        return FittedNeuralField(
            config=fitted.config,
            field=fitted.field,
            params=fitted.params,
            losses=fitted.losses,
            has_sigma=fitted.has_sigma,
            xarray_coords=tuple(coords),
            time_origins=origins,
        )

    def fit_xarray(
    self,
    ds: Any,
    *,
    coords: Sequence[str],
    target: str,
    sigma: str | None = None,
    time_origin: Any = None,
    seed: PRNGKeyArray | int,
) -> FittedNeuralField:
    r"""Fit on an xarray Dataset (``pip install pyrox-nn[xarray]``).

    Every variable named in ``coords``, ``target`` and ``sigma`` (a
    dimension, coordinate or data variable) is broadcast against the
    others and flattened, in the target's dimension order, to one row
    per grid point; ``coords`` become the columns of ``X`` in the order
    given. Rows where any of them is missing (NaN, NaT) are masked out,
    so a gappy grid needs no reshaping.

    A datetime coordinate becomes float days since ``time_origin``
    (default: its earliest timestamp), so the column starts near 0
    instead of at ~2·10⁴ days since 1970, which float32 would resolve
    only to minutes and a SIREN would see as a huge frequency. A
    timedelta coordinate becomes float days. The origins are stored on
    the fit; `FittedNeuralField.predict_xarray` applies the same
    mapping, and ``predict`` on arrays expects the same units.

    Args:
        ds: An ``xarray.Dataset``.
        coords: Names of the input variables, e.g. ``("lon", "lat", "time")``.
        target: Name of the target variable.
        sigma: Optional name of the per-row noise std variable.
        time_origin: Optional reference time for every datetime
            coordinate (anything ``numpy.datetime64`` accepts).
        seed: PRNG key or integer seed.

    Returns:
        The fitted ensemble, which remembers ``coords`` and the time
        origins.

    Raises:
        ImportError: If xarray is not installed.
        ValueError: If a name is not a variable of ``ds``, or as `fit`.
    """
    xr = _require_xarray("fit_xarray")
    names = [*coords, target, *([sigma] if sigma is not None else [])]
    _check_names(ds, names)
    # Flatten in the target's dimension order (C order), whatever the
    # order of ``coords``.
    arrays = xr.broadcast(*(ds[name] for name in names))
    dims = (
        *ds[target].dims,
        *(d for d in arrays[0].dims if d not in ds[target].dims),
    )
    arrays = [a.transpose(*dims) for a in arrays]
    origins = tuple(_time_origin(ds[name].values, time_origin) for name in coords)
    cols = [
        _numeric_column(a.values.ravel(), origin)
        for a, origin in zip(
            arrays, (*origins, *([None] * (len(names) - len(coords)))), strict=True
        )
    ]
    X = np.stack(cols[: len(coords)], axis=-1)
    valid = np.isfinite(np.stack(cols, axis=-1)).all(axis=-1)
    fitted = self.fit(
        X,
        cols[len(coords)],
        sigma=None if sigma is None else cols[-1],
        valid=valid,
        seed=seed,
    )
    return FittedNeuralField(
        config=fitted.config,
        field=fitted.field,
        params=fitted.params,
        losses=fitted.losses,
        has_sigma=fitted.has_sigma,
        xarray_coords=tuple(coords),
        time_origins=origins,
    )

    def predict(
        self,
        X: Float[ArrayLike, "N D"],
        *,
        quantiles: Sequence[float] | None = None,

        ``num_posterior_samples`` guide draws each. The quantiles are those

        Returns:
            The mean of shape ``(N,)``, or ``(mean, quantiles)``.

        return _predict(
            self.params,
            self.field,
            _floating(X),
            encoder=cfg.encoder,
            noise=cfg.noise,
            sigma_obs=cfg.sigma_obs,
            has_sigma=self.has_sigma,
            inference=cfg.inference,
            guide_factory=cfg.guide_factory,
            quantiles=quantiles,
            predict_seed=predict_seed,

    def predict_xarray(
        self,
        ds: Any,
        *,
        quantiles: Sequence[float] | None = None,
        predict_seed: int = 0,
        num_posterior_samples: int = 32,
    ) -> Any:
        """Predict on the grid of an xarray Dataset, as fitted by ``fit_xarray``.

        The fitted ``coords`` are read from ``ds``, broadcast against each
        other and mapped to the fit's units (datetimes as days since the
        stored origins); the result has their broadcast shape and
        coordinates. A grid point with a missing coordinate predicts NaN.

        Args:
            ds: An ``xarray.Dataset`` holding the fitted ``coords``.
            quantiles: Optional levels, as in `predict`.
            predict_seed: As in `predict`.
            num_posterior_samples: As in `predict`.

        Returns:
            The mean as an ``xarray.DataArray``, or ``(mean, quantiles)``
            with one ``DataArray`` per level.

        Raises:
            ValueError: If this fit did not come from ``fit_xarray``, or a
                coordinate is not a variable of ``ds``.
        """
        if self.xarray_coords is None or self.time_origins is None:
            raise ValueError("predict_xarray needs a fit from fit_xarray.")
        xr = _require_xarray("predict_xarray")
        _check_names(ds, self.xarray_coords)
        arrays = xr.broadcast(*(ds[name] for name in self.xarray_coords))
        template = arrays[0]
        cols = [
            _numeric_column(a.transpose(*template.dims).values.ravel(), origin)
            for a, origin in zip(arrays, self.time_origins, strict=True)
        ]
        X = np.stack(cols, axis=-1)
        rows = np.asarray(np.isfinite(X).all(axis=-1))
        out = self.predict(
            np.where(rows[:, None], X, 0.0),
            quantiles=quantiles,
            predict_seed=predict_seed,
            num_posterior_samples=num_posterior_samples,
        )

        def as_grid(values: Any) -> Any:
            grid = np.where(rows, np.asarray(values), np.nan).reshape(template.shape)
            return xr.DataArray(grid, coords=template.coords, dims=template.dims)

        if quantiles is None:
            return as_grid(out)
        mean, qs = out
        return as_grid(mean), tuple(as_grid(q) for q in qs)
