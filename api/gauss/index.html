    Gauss API¶

    pyrox_gauss holds the Gaussian NumPyro distributions whose covariance or precision is a gaussx operator. They moved here from gaussx with their names, constructors and numerics unchanged (#453), so moving to them is an import change. pyrox_gauss imports gaussx and NumPyro and never imports pyrox; the model packages build these distributions from the hyperparameters they sample.

    Multivariate normals¶

    AbstractMultivariateNormal ¶

    Bases: Distribution

    Common surface of MultivariateNormal and MultivariateNormalPrecision.

    x ~ 𝒩(μ, Σ),   Λ = Σ⁻¹

    A subclass stores one native operator: the covariance Σ (MultivariateNormal) or the precision Λ (MultivariateNormalPrecision). The other one is the lazy gaussx.inv of it. Code written against this base works for either parameterisation:

    • covariance_operator / precision_operator: lineax operators, one native and one a lazy inverse.
    • covariance_matrix / precision_matrix / scale_tril: dense arrays with numpyro's names and batch broadcasting. They materialise D_x × D_x matrices, so they are for small problems and interop.
    • mean, log_prob, kl.

    Requires the numpyro optional extra (pip install "gaussx[numpyro]").

    Examples:

    Both parameterisations expose the same accessors:

    >>> import jax.numpy as jnp
>>> import lineax as lx
>>> from pyrox_gauss import (
...     AbstractMultivariateNormal,
...     MultivariateNormal,
...     MultivariateNormalPrecision,
... )
>>> covariance = lx.MatrixLinearOperator(
...     2.0 * jnp.eye(2), lx.positive_semidefinite_tag
... )
>>> precision = lx.MatrixLinearOperator(
...     0.5 * jnp.eye(2), lx.positive_semidefinite_tag
... )
>>> p = MultivariateNormal(jnp.zeros(2), covariance)
>>> q = MultivariateNormalPrecision(jnp.zeros(2), precision)
>>> isinstance(q, AbstractMultivariateNormal)
True
>>> bool(jnp.allclose(p.precision_matrix, q.precision_matrix))
True

    Equal Gaussians have zero KL, whatever the parameterisation:

    >>> bool(jnp.allclose(p.kl(q), 0.0, atol=1e-6))
True

    Batched means broadcast the dense accessors:

    >>> batched = MultivariateNormal(jnp.zeros((3, 2)), covariance)
>>> batched.covariance_matrix.shape, batched.log_prob(jnp.ones(2)).shape
((3, 2, 2), (3,))

    class AbstractMultivariateNormal(dist.Distribution):
    """Common surface of `MultivariateNormal` and `MultivariateNormalPrecision`.

        x ~ 𝒩(μ, Σ),   Λ = Σ⁻¹

    A subclass stores one native operator: the covariance Σ
    (`MultivariateNormal`) or the precision Λ (`MultivariateNormalPrecision`).
    The other one is the lazy `gaussx.inv` of it. Code written against this
    base works for either parameterisation:

    - ``covariance_operator`` / ``precision_operator``: lineax operators,
      one native and one a lazy inverse.
    - ``covariance_matrix`` / ``precision_matrix`` / ``scale_tril``: dense
      arrays with numpyro's names and batch broadcasting. They materialise
      D_x × D_x matrices, so they are for small problems and interop.
    - ``mean``, ``log_prob``, `kl`.

    Requires the ``numpyro`` optional extra
    (``pip install "gaussx[numpyro]"``).

    Examples:
        Both parameterisations expose the same accessors:

        >>> import jax.numpy as jnp
        >>> import lineax as lx
        >>> from pyrox_gauss import (
        ...     AbstractMultivariateNormal,
        ...     MultivariateNormal,
        ...     MultivariateNormalPrecision,
        ... )
        >>> covariance = lx.MatrixLinearOperator(
        ...     2.0 * jnp.eye(2), lx.positive_semidefinite_tag
        ... )
        >>> precision = lx.MatrixLinearOperator(
        ...     0.5 * jnp.eye(2), lx.positive_semidefinite_tag
        ... )
        >>> p = MultivariateNormal(jnp.zeros(2), covariance)
        >>> q = MultivariateNormalPrecision(jnp.zeros(2), precision)
        >>> isinstance(q, AbstractMultivariateNormal)
        True
        >>> bool(jnp.allclose(p.precision_matrix, q.precision_matrix))
        True

        Equal Gaussians have zero KL, whatever the parameterisation:

        >>> bool(jnp.allclose(p.kl(q), 0.0, atol=1e-6))
        True

        Batched means broadcast the dense accessors:

        >>> batched = MultivariateNormal(jnp.zeros((3, 2)), covariance)
        >>> batched.covariance_matrix.shape, batched.log_prob(jnp.ones(2)).shape
        ((3, 2, 2), (3,))
    """

    arg_constraints = {"loc": dist.constraints.real_vector}  # noqa: RUF012
    support = dist.constraints.real_vector
    reparametrized_params = ["loc"]  # noqa: RUF012

    loc: Float[Array, "*batch N"]
    solver: AbstractSolverStrategy

    @property
    def covariance_operator(self) -> lx.AbstractLinearOperator:
        """The covariance Σ as a lineax operator."""
        raise NotImplementedError

    @property
    def precision_operator(self) -> lx.AbstractLinearOperator:
        """The precision Λ = Σ⁻¹ as a lineax operator."""
        raise NotImplementedError

    def _log_prob_single(self, residual: Float[Array, " N"]) -> Float[Array, ""]:
        """Log-density of one event from its residual r = x − μ.

        Args:
            residual: Residual r = x − μ of a single event.

        Returns:
            The scalar log 𝒩(x; μ, Σ).
        """
        raise NotImplementedError

    @validate_sample
    def log_prob(self, value: Float[Array, "*batch N"]) -> Float[Array, "*batch"]:
        """Log-density log 𝒩(x; μ, Σ), broadcast over the batch shape.

            log 𝒩(x; μ, Σ) = −½ (D_x log 2π + log det Σ + rᵀ Σ⁻¹ r),
            r = x − μ

        Algorithm:
            1. Form the residuals r = x − μ, broadcasting x against μ.
            2. Flatten the leading axes and evaluate one event at a time.
            3. Restore the leading shape.

        Args:
            value: Points x, with any leading shape that broadcasts against
                the batch shape.

        Returns:
            One log-density per broadcast batch element.
        """
        # r = x − μ
        residual = value - self.loc
        leading_shape = residual.shape[:-1]
        residual_flat = rearrange(residual, "... D -> (...) D")
        log_prob_flat = jax.vmap(self._log_prob_single)(residual_flat)
        return _reshape_batch(log_prob_flat, leading_shape)

    @lazy_property
    def mean(self) -> Float[Array, "*batch N"]:
        """The mean μ."""
        return self.loc

    def _broadcast_matrix(self, matrix: Float[Array, "N N"]) -> Array:
        """Broadcast a shared D_x × D_x matrix over the batch shape.

        Args:
            matrix: The shared matrix.

        Returns:
            The matrix repeated for every batch element.
        """
        return jnp.broadcast_to(matrix, self.batch_shape + self.event_shape * 2)

    @lazy_property
    def covariance_matrix(self) -> Float[Array, "*batch N N"]:
        """Dense Σ, broadcast over the batch shape."""
        return self._broadcast_matrix(self.covariance_operator.as_matrix())

    @lazy_property
    def precision_matrix(self) -> Float[Array, "*batch N N"]:
        """Dense Λ = Σ⁻¹, broadcast over the batch shape."""
        return self._broadcast_matrix(self.precision_operator.as_matrix())

    @lazy_property
    def scale_tril(self) -> Float[Array, "*batch N N"]:
        """Dense lower Cholesky factor L of Σ = L Lᵀ (numpyro's name)."""
        # L Lᵀ = Σ
        return jnp.linalg.cholesky(self.covariance_matrix)

    def kl(self, other: AbstractMultivariateNormal) -> Float[Array, "*batch"]:
        """KL(first ‖ second): KL[self ‖ other] in closed form.

            KL[p ‖ q] = ½ (tr(Σ_q⁻¹ Σ_p) + (μ_q − μ_p)ᵀ Σ_q⁻¹ (μ_q − μ_p) − D_x
                           + log det Σ_q − log det Σ_p)

        with p = self and q = other. Any mix of covariance and precision
        parameterisations works. A precision-parameterised ``other`` needs no
        solve for tr(Σ_q⁻¹ Σ_p) or the quadratic term, because the inverse of
        its lazy covariance is its precision. Batch shapes broadcast as in
        numpyro's ``kl_divergence``. Unlike
        ``numpyro.distributions.kl_divergence``, it uses the closed form
        whatever the solver strategies are.

        Args:
            other: The second Gaussian q, with the same event shape.

        Returns:
            The KL divergence, with the broadcast batch shape.
        """
        # lazy import, cycle: _numpyro_kl -> _mvn_base
        from pyrox_gauss._numpyro_kl import closed_form_kl

        return closed_form_kl(self, other)

    MultivariateNormal ¶

    Bases: AbstractMultivariateNormal

    Multivariate normal 𝒩(μ, Σ) whose covariance Σ is a lineax operator.

    x ~ 𝒩(μ, Σ)
log 𝒩(x; μ, Σ) = −½ (D_x log 2π + log det Σ + (x − μ)ᵀ Σ⁻¹ (x − μ))

    Covariance-parameterised member of AbstractMultivariateNormal, which supplies the shared accessors (covariance_operator, precision_operator, covariance_matrix, scale_tril, kl, …). Unlike numpyro.distributions.MultivariateNormal, which needs dense arrays, it accepts any lineax.AbstractLinearOperator as Σ, so the log-density, sampling and entropy use gaussx structural dispatch for structured covariances (Kronecker, block-diagonal, low-rank, diagonal, …).

    Requires the numpyro optional extra (pip install "gaussx[numpyro]").

    Parameters:

    Name Type Description Default
    loc Float[Array, '*batch N']

    Mean μ; leading axes are the batch shape.

    		 required
    cov_operator AbstractLinearOperator

    Covariance Σ, shared by every batch element.

    		 required
    solver AbstractSolverStrategy | None

    Solver strategy for the solves and log-determinants of Σ. Defaults to AutoSolver().

    		 None
    validate_args bool | None

    Whether to validate input arguments.

    		 None

    Examples:

    A standard normal in three dimensions:

    >>> import jax
>>> import jax.numpy as jnp
>>> import lineax as lx
>>> from pyrox_gauss import MultivariateNormal
>>> covariance = lx.MatrixLinearOperator(
...     jnp.eye(3), lx.positive_semidefinite_tag

    """Multivariate normal 𝒩(μ, Σ) whose covariance Σ is a lineax operator.

        x ~ 𝒩(μ, Σ)
        log 𝒩(x; μ, Σ) = −½ (D_x log 2π + log det Σ + (x − μ)ᵀ Σ⁻¹ (x − μ))

    Covariance-parameterised member of `AbstractMultivariateNormal`, which
    supplies the shared accessors (``covariance_operator``,
    ``precision_operator``, ``covariance_matrix``, ``scale_tril``, `kl`, …).
    Unlike ``numpyro.distributions.MultivariateNormal``, which needs dense
    arrays, it accepts any ``lineax.AbstractLinearOperator`` as Σ, so the
    log-density, sampling and entropy use gaussx structural dispatch for
    structured covariances (Kronecker, block-diagonal, low-rank, diagonal,
    …).

    Requires the ``numpyro`` optional extra
    (``pip install "gaussx[numpyro]"``).

    Args:

        cov_operator: Covariance Σ, shared by every batch element.

            Defaults to ``AutoSolver()``.
        validate_args: Whether to validate input arguments.

        >>> import jax

        x ~ 𝒩(μ, Λ⁻¹)
        log 𝒩(x; μ, Λ⁻¹) = −½ (D_x log 2π − log det Λ + (x − μ)ᵀ Λ (x − μ))

        loc: Mean μ; leading axes are the batch shape.
        prec_operator: Precision Λ = Σ⁻¹, shared by every batch element.
        solver: Solver strategy for the log-determinant of Λ. Defaults to

        self,
        loc: Float[Array, "*batch N"],
        prec_operator: lx.AbstractLinearOperator,
        solver: AbstractSolverStrategy | None = None,
        *,
        validate_args: bool | None = None,
    ) -> None:

            solver = AutoSolver()
        self.loc = loc
        self.prec_operator = prec_operator
        self.solver = solver
        event_shape = loc.shape[-1:]
        batch_shape = loc.shape[:-1]

    def _log_prob_single(self, residual: Float[Array, " N"]) -> Float[Array, ""]:
        """Log-density of one event from its residual r = x − μ.

            log 𝒩(x; μ, Λ⁻¹) = −½ (D_x log 2π − log det Λ + rᵀ Λ r)

        Args:
            residual: Residual r = x − μ of a single event.

        Returns:
            The scalar log 𝒩(x; μ, Λ⁻¹).
        """
        # Λ r
        precision_residual = self.prec_operator.mv(residual)
        # rᵀ Λ r, a full reduction of one event's vector
        mahalanobis = jnp.sum(residual * precision_residual)
        # log det Λ
        logdet = self.solver.logdet(self.prec_operator)
        event_size = self.loc.shape[-1]
        # log 𝒩(x; μ, Λ⁻¹) = −½ (D_x log 2π − log det Λ + rᵀ Λ r)
        return -0.5 * (event_size * _LOG_2PI - logdet + mahalanobis)

    def sample(
        self,
        key: jax.Array | None,
        sample_shape: tuple[int, ...] = (),
    ) -> Float[Array, "*batch N"]:
        """Draw x = μ + L_Λ⁻ᵀ z with z ~ 𝒩(0, I), by a triangular solve.

            x = μ + L_Λ⁻ᵀ z,   z ~ 𝒩(0, I),   Λ = L_Λ L_Λᵀ

        Algorithm:
            1. Draw z ~ 𝒩(0, I) in the parameters' dtype.
            2. Strip tags and scalar multiples, Λ = c Λ₀, so a structured Λ₀
               keeps its structured Cholesky factor L_Λ₀.
            3. Solve L_Λ₀ᵀ x₀ = z and rescale, x − μ = c^(−½) x₀.
            4. For a dense Λ whose Cholesky fails (non-finite draws), use
               the symmetric inverse square root x − μ = Λ^(−½) z instead.
            5. Add the mean μ.

        Args:
            key: PRNG key.
            sample_shape: Leading shape of the draws.

        Returns:
            Draws of shape ``sample_shape + batch_shape + event_shape``.

        Raises:
            ValueError: If ``key`` is ``None``.
        """
        if key is None:
            raise ValueError(
                "PRNG key must be provided to sample from MultivariateNormalPrecision."
            )
        shape = sample_shape + self.batch_shape + self.event_shape
        # Draw in the parameters' dtype: default-float noise meets a float32
        # factor in the solve below, which lineax rejects under x64.
        dtype = jnp.result_type(self.loc, self.prec_operator.in_structure().dtype)
        # z ~ 𝒩(0, I)
        noise = jax.random.normal(key, shape=shape, dtype=dtype)  # type: ignore[arg-type]
        noise_flat = rearrange(noise, "... D -> (...) D")

        # Look through tags and scalar multiples so a structured precision
        # (e.g. 2 * Kronecker) keeps its structured Cholesky factor.
        # Λ = c Λ₀, s = c^(−½)
        precision, scale = _unwrap_scaled(self.prec_operator)
        # L_Λ₀ L_Λ₀ᵀ = Λ₀
        factor = _cholesky(precision)

        def _solve_one(standard_normal):
            if isinstance(factor, SparseCholeskyFactor):
                # A SparseOperator's factor is of P Λ₀ Pᵀ: x₀ = Pᵀ L⁻ᵀ z.
                return factor.solve_lower_transpose(standard_normal)
            # x₀ = L_Λ₀⁻ᵀ z
            return _solve(factor.T, standard_normal)

        # L_Λ₀⁻ᵀ z, one row per draw
        unscaled_flat = jax.vmap(_solve_one)(noise_flat)
        # x − μ = s L_Λ₀⁻ᵀ z
        draws_flat = scale * unscaled_flat
        if isinstance(precision, lx.MatrixLinearOperator):
            # A dense Cholesky that fails numerically (an ill-conditioned or
            # only semi-definite precision) gives non-finite draws; take the
            # dense symmetric inverse square root then. Dense precisions only:
            # lax.cond stages both branches under jit, and a dense branch
            # would put D_x × D_x buffers into every structured or sparse
            # sample.
            structured = draws_flat
            draws_flat = jax.lax.cond(
                jnp.all(jnp.isfinite(structured)),
                lambda: structured,
                lambda: _dense_precision_draws(self.prec_operator, noise_flat),
            )
        centred = _reshape_samples(draws_flat, shape[:-1])
        # x = μ + (x − μ)
        return self.loc + centred

    @lazy_property
    def variance(self) -> Float[Array, "*batch N"]:
        """Marginal variances diag(Λ⁻¹), broadcast over the batch shape.

        The exact structured paths of `gaussx.diag_inv` (e.g. the O(N d³)
        selected inverse of a `gaussx.BlockTriDiag`) apply; only an operator
        with none of them densifies (gh-360).

        Returns:

    self,
    key: jax.Array | None,
    sample_shape: tuple[int, ...] = (),
) -> Float[Array, "*batch N"]:
    """Draw x = μ + L_Λ⁻ᵀ z with z ~ 𝒩(0, I), by a triangular solve.

        x = μ + L_Λ⁻ᵀ z,   z ~ 𝒩(0, I),   Λ = L_Λ L_Λᵀ

    Algorithm:
        1. Draw z ~ 𝒩(0, I) in the parameters' dtype.
        2. Strip tags and scalar multiples, Λ = c Λ₀, so a structured Λ₀
           keeps its structured Cholesky factor L_Λ₀.
        3. Solve L_Λ₀ᵀ x₀ = z and rescale, x − μ = c^(−½) x₀.
        4. For a dense Λ whose Cholesky fails (non-finite draws), use
           the symmetric inverse square root x − μ = Λ^(−½) z instead.
        5. Add the mean μ.

    Args:
        key: PRNG key.
        sample_shape: Leading shape of the draws.

    Returns:
        Draws of shape ``sample_shape + batch_shape + event_shape``.

    Raises:
        ValueError: If ``key`` is ``None``.
    """
    if key is None:
        raise ValueError(
            "PRNG key must be provided to sample from MultivariateNormalPrecision."
        )
    shape = sample_shape + self.batch_shape + self.event_shape
    # Draw in the parameters' dtype: default-float noise meets a float32
    # factor in the solve below, which lineax rejects under x64.
    dtype = jnp.result_type(self.loc, self.prec_operator.in_structure().dtype)
    # z ~ 𝒩(0, I)
    noise = jax.random.normal(key, shape=shape, dtype=dtype)  # type: ignore[arg-type]
    noise_flat = rearrange(noise, "... D -> (...) D")

    # Look through tags and scalar multiples so a structured precision
    # (e.g. 2 * Kronecker) keeps its structured Cholesky factor.
    # Λ = c Λ₀, s = c^(−½)
    precision, scale = _unwrap_scaled(self.prec_operator)
    # L_Λ₀ L_Λ₀ᵀ = Λ₀
    factor = _cholesky(precision)

    def _solve_one(standard_normal):
        if isinstance(factor, SparseCholeskyFactor):
            # A SparseOperator's factor is of P Λ₀ Pᵀ: x₀ = Pᵀ L⁻ᵀ z.
            return factor.solve_lower_transpose(standard_normal)
        # x₀ = L_Λ₀⁻ᵀ z
        return _solve(factor.T, standard_normal)

    # L_Λ₀⁻ᵀ z, one row per draw
    unscaled_flat = jax.vmap(_solve_one)(noise_flat)
    # x − μ = s L_Λ₀⁻ᵀ z
    draws_flat = scale * unscaled_flat
    if isinstance(precision, lx.MatrixLinearOperator):
        # A dense Cholesky that fails numerically (an ill-conditioned or
        # only semi-definite precision) gives non-finite draws; take the
        # dense symmetric inverse square root then. Dense precisions only:
        # lax.cond stages both branches under jit, and a dense branch
        # would put D_x × D_x buffers into every structured or sparse
        # sample.
        structured = draws_flat
        draws_flat = jax.lax.cond(
            jnp.all(jnp.isfinite(structured)),
            lambda: structured,
            lambda: _dense_precision_draws(self.prec_operator, noise_flat),
        )
    centred = _reshape_samples(draws_flat, shape[:-1])
    # x = μ + (x − μ)
    return self.loc + centred

    def _sample_single(self, key: jax.Array) -> Float[Array, "T M"]:
        """One ancestral forward simulation of the series y_1:T.

            x₀ = μ_x,0 + L_x,0 z₀
            x_t = F x_t−1 + L_η z_η,t
            y_t = H x_t + L_ε z_ε,t

        Cholesky factors go through the structural `gaussx.cholesky`
        primitive, so a Kronecker, block-diagonal or diagonal covariance is
        never materialised.

        Args:
            key: PRNG key.

        Returns:
            The observations y_1:T, one row per step.
        """
        state_size = self.m0.shape[0]
        obs_size = self.event_shape[1]
        prior_key, model_noise_key, obs_noise_key = jax.random.split(key, 3)
        # L_x,0 L_x,0ᵀ = Σ_x,0
        prior_factor = _cholesky(_psd(self.P0))
        # L_η L_ηᵀ = Σ_η
        model_noise_factor = _cholesky(_psd(self.Q))
        # L_ε L_εᵀ = Σ_ε
        obs_noise_factor = _cholesky(_psd(self.R))

        # Each noise term is drawn in its factor's dtype, not JAX's default
        # float, so a float32 model stays float32 under x64.
        # z₀ ~ 𝒩(0, I)
        prior_noise = jax.random.normal(
            prior_key, (state_size,), dtype=prior_factor.in_structure().dtype
        )
        # L_x,0 z₀
        prior_deviation = prior_factor.mv(prior_noise)
        # x₀ = μ_x,0 + L_x,0 z₀
        initial_state = self.m0 + prior_deviation
        # z_η,t ~ 𝒩(0, I), one per step
        model_noise_draws = jax.random.normal(
            model_noise_key,
            (self.n_steps, state_size),
            dtype=model_noise_factor.in_structure().dtype,
        )
        # z_ε,t ~ 𝒩(0, I), one per step
        obs_noise_draws = jax.random.normal(
            obs_noise_key,
            (self.n_steps, obs_size),
            dtype=obs_noise_factor.in_structure().dtype,
        )

        def step(state, noise):
            model_noise, obs_noise = noise
            # F x_t−1
            propagated = _matvec(self.A, state)
            # L_η z_η,t
            model_deviation = model_noise_factor.mv(model_noise)
            # x_t = F x_t−1 + L_η z_η,t
            state_next = propagated + model_deviation
            # H x_t
            predicted_obs = _matvec(self.H, state_next)
            # L_ε z_ε,t
            obs_deviation = obs_noise_factor.mv(obs_noise)
            # y_t = H x_t + L_ε z_ε,t
            obs = predicted_obs + obs_deviation
            return state_next, obs

        _, observations = jax.lax.scan(
            step, initial_state, (model_noise_draws, obs_noise_draws)
        )
        return observations

    def sample(
        self,
        key: jax.Array | None,
        sample_shape: tuple[int, ...] = (),
    ) -> Float[Array, "*sample T M"]:
        """Ancestral draws of the observation series y_1:T.

        Args:
            key: PRNG key.
            sample_shape: Leading shape of the draws.

        Returns:
            Draws of shape ``sample_shape + (T, M)``.

        Raises:
            ValueError: If ``key`` is ``None``.
        """
        if key is None:
            raise ValueError("PRNG key must be provided to sample from LGSSM.")

    Gauss-Markov chain over the states x_0:T−1 as a density over (T, d).

    x₀ ~ 𝒩(μ_x,0, Σ_x,0)
x_t+1 = F_t x_t + c_t + η_t+1,   η_t+1 ~ 𝒩(0, Σ_η,t+1)

    for t = 0, …, T − 2 and states x_t ∈ ℝ^{D_x}.     class MarkovGaussian(dist.Distribution):
    """Gauss-Markov chain over the states x_0:T−1 as a density over ``(T, d)``.

        x₀ ~ 𝒩(μ_x,0, Σ_x,0)
        x_t+1 = F_t x_t + c_t + η_t+1,   η_t+1 ~ 𝒩(0, Σ_η,t+1)

    for t = 0, …, T − 2 and states x_t ∈ ℝ^{D_x}. This generative
    state-space form is the canonical parameterisation; the joint
    covariance Σ_x of x_0:T−1 has a block-tridiagonal inverse Λ_x.
    ``event_shape`` is ``(T, d)``, so ``log_prob`` of a trajectory is a
    scalar and the distribution can be used directly as a NumPyro site: a
    prior over a latent path, or a structured variational guide. Every view
    is derived from the state-space parameters on demand:

    | View | Cost | Route |
    |---|---|---|
    | `marginals`, `mean`, `variance` | O(T D_x³) | forward moment propagation |
    | `sample`, `log_prob` | O(T D_x³) | chain factorisation, no joint Cholesky |
    | `precision`, `to_precision_form` | O(T D_x³) | `udl_from_ssm_params` |
    | `from_precision_form` | O(T D_x³) | `udl_decomposition` |
    | `covariance_matrix` | O(T² D_x³) | banded solves against I; small T |

    Where `LGSSM` is a density over the *observations* of a chain with
    the states marginalised out, this is the density over the *states*
    themselves. Requires the ``numpyro`` optional extra
    (``pip install "gaussx[numpyro]"``).

    Args:
        A: Transition matrices F_t, with ``A[t]`` mapping x_t to x_t+1.
        Q: Model-noise covariances Σ_η,t+1, with ``Q[t]`` driving the step
            from x_t to x_t+1.
        mu0: Prior mean μ_x,0 of the initial state x₀.
        P0: Prior covariance Σ_x,0 of x₀.
        b: Optional transition offsets c_t, with ``b[t]`` added in the step
            from x_t to x_t+1. Defaults to zero.
        validate_args: Whether to validate input arguments.

    Raises:
        ValueError: If ``A`` is not a stack of square matrices, or ``Q``,
            ``mu0``, ``P0`` or ``b`` does not match its shape.

    Examples:
        A ten-step chain in two dimensions:

        >>> import jax
        >>> import jax.numpy as jnp
        >>> import gaussx
        >>> import pyrox_gauss
        >>> T, d = 10, 2
        >>> A = jnp.broadcast_to(0.9 * jnp.eye(d), (T - 1, d, d))
        >>> Q = jnp.broadcast_to(0.1 * jnp.eye(d), (T - 1, d, d))
        >>> chain = pyrox_gauss.MarkovGaussian(A, Q, jnp.zeros(d), jnp.eye(d))
        >>> xs = chain.sample(jax.random.key(0), (4,))
        >>> xs.shape, chain.log_prob(xs).shape
        ((4, 10, 2), (4,))

        The precision form round-trips to the same chain:

        >>> mean, precision = chain.to_precision_form()
        >>> mean.shape, precision.as_matrix().shape
        ((20,), (20, 20))
        >>> rebuilt = pyrox_gauss.MarkovGaussian.from_precision_form(mean, precision)
        >>> bool(jnp.allclose(rebuilt.A, A, atol=1e-4))
        True

        A single state has no transitions:

        >>> single = pyrox_gauss.MarkovGaussian(A[:0], Q[:0], jnp.zeros(d), jnp.eye(d))
        >>> single.event_shape, single.cross_covariances().shape
        ((1, 2), (0, 2, 2))
    """

    arg_constraints = {}  # noqa: RUF012
    support = dist.constraints.real_matrix
    reparametrized_params = ["A", "b", "Q", "mu0", "P0"]  # noqa: RUF012
    pytree_data_fields = ("A", "b", "Q", "mu0", "P0")

    def __init__(
        self,
        A: Float[Array, "Tm1 d d"],
        Q: Float[Array, "Tm1 d d"],
        mu0: Float[Array, " d"],
        P0: Float[Array, "d d"],
        *,
        b: Float[Array, "Tm1 d"] | None = None,
        validate_args: bool | None = None,
    ) -> None:
        if A.ndim != 3 or A.shape[1] != A.shape[2]:
            raise ValueError(f"A must have shape (T-1, d, d), got {A.shape}.")
        Tm1, d, _ = A.shape
        if Q.shape != (Tm1, d, d):
            raise ValueError(f"Q must have shape {(Tm1, d, d)}, got {Q.shape}.")
        if mu0.shape != (d,):
            raise ValueError(f"mu0 must have shape {(d,)}, got {mu0.shape}.")
        if P0.shape != (d, d):
            raise ValueError(f"P0 must have shape {(d, d)}, got {P0.shape}.")
        if b is None:
            b = jnp.zeros((Tm1, d), dtype=A.dtype)
        elif b.shape != (Tm1, d):
            raise ValueError(f"b must have shape {(Tm1, d)}, got {b.shape}.")
        self.A = A
        self.b = b
        self.Q = Q
        self.mu0 = mu0
        self.P0 = P0
        super().__init__(
            batch_shape=(),
            event_shape=(Tm1 + 1, d),
            validate_args=validate_args,
        )

    @property
    def horizon(self) -> int:
        """Number of states T in the chain."""
        return self.event_shape[0]

    @property
    def state_dim(self) -> int:
        """State size D_x."""
        return self.event_shape[1]

    # ------------------------------------------------------------------
    # Moment views
    # ------------------------------------------------------------------

    def marginals(self) -> tuple[Float[Array, "T d"], Float[Array, "T d d"]]:
        """Marginal means μ_x,t and covariances Σ_x,t of every state.

            μ_x,t+1 = F_t μ_x,t + c_t
            Σ_x,t+1 = F_t Σ_x,t F_tᵀ + Σ_η,t+1

        Algorithm:
            1. Start from the prior moments μ_x,0 and Σ_x,0.
            2. Push the moments through each transition with a scan.
            3. Prepend the prior moments to the propagated ones.

        Returns:
            Tuple ``(means, covs)`` holding μ_x,t and Σ_x,t for
            t = 0, …, T − 1.
        """

        def _step(carry, inputs):
            mean, covariance = carry
            transition, offset, model_noise_covariance = inputs
            # F_t μ_x,t
            propagated_mean = transition @ mean
            # μ_x,t+1 = F_t μ_x,t + c_t
            mean_next = propagated_mean + offset
            # F_t Σ_x,t F_tᵀ, one three-operand contraction
            propagated_covariance = einsum(
                transition,
                covariance,
                transition,
                "next prev, prev col, other col -> next other",
            )
            # Σ_x,t+1 = F_t Σ_x,t F_tᵀ + Σ_η,t+1
            covariance_next = propagated_covariance + model_noise_covariance
            return (mean_next, covariance_next), (mean_next, covariance_next)

        _, (means_rest, covs_rest) = jax.lax.scan(
            _step, (self.mu0, self.P0), (self.A, self.b, self.Q)
        )
        # μ_x,0 as the first row
        prior_mean_row = rearrange(self.mu0, "state -> 1 state")
        # Σ_x,0 as the first slab
        prior_covariance_slab = rearrange(self.P0, "row col -> 1 row col")
        # μ_x,0:T−1, prior first
        means = jnp.concatenate([prior_mean_row, means_rest])
        # Σ_x,0:T−1, prior first
        covs = jnp.concatenate([prior_covariance_slab, covs_rest])
        return means, covs

    @lazy_property
    def mean(self) -> Float[Array, "T d"]:
        """Marginal means μ_x,t, one row per state."""
        return self.marginals()[0]

    @lazy_property
    def variance(self) -> Float[Array, "T d"]:
        """Marginal variances diag(Σ_x,t), one row per state."""
        _, covs = self.marginals()
        # diag(Σ_x,t)
        return rearrange(covs, "step state state -> step state")

    def cross_covariances(self) -> Float[Array, "Tm1 d d"]:
        """Lag-one cross-covariances Σ_x,t+1,t = Cov[x_t+1, x_t] = F_t Σ_x,t.

        Returns:
            One D_x × D_x block per transition, t = 0, …, T − 2.
        """
        _, covs = self.marginals()
        return self._cross(covs)

    def _cross(self, covs: Float[Array, "T d d"]) -> Float[Array, "Tm1 d d"]:
        """Lag-one cross-covariances Σ_x,t+1,t = F_t Σ_x,t from the marginals.

        Args:
            covs: Marginal covariances Σ_x,t, one per state.

        Returns:
            One block Σ_x,t+1,t per transition.
        """
        if self.A.shape[0] == 0:
            # One step, no transitions: einx cannot contract the empty
            # transition axis (gh-348).
            return self.A
        # Σ_x,t+1,t = F_t Σ_x,t
        return einsum(
            self.A, covs[:-1], "step next prev, step prev col -> step next col"
        )

    def pairwise_marginals(
        self,
    ) -> tuple[Float[Array, "Tm1 two_d"], Float[Array, "Tm1 two_d two_d"]]:
        """Joint Gaussian p(x_t, x_t+1) of each pair of consecutive states.

            p(x_t, x_t+1) = 𝒩([μ_x,t; μ_x,t+1], [[Σ_x,t, Σ_x,t+1,tᵀ],
                                                [Σ_x,t+1,t, Σ_x,t+1]])

        Returns:
            Tuple ``(joint_means, joint_covs)`` with one pair per
            transition, each of length 2 D_x, in the layout of
            `gaussx.pairwise_marginals`.
        """
        means, covs = self.marginals()
        if self.A.shape[0] == 0:
            # One state, no pairs.
            # 2 D_x
            joint_size = 2 * self.state_dim
            return (
                jnp.zeros((0, joint_size), means.dtype),
                jnp.zeros((0, joint_size, joint_size), covs.dtype),
            )
        return pairwise_marginals(means, covs, self._cross(covs))

    @lazy_property
    def covariance_matrix(self) -> Float[Array, "Td Td"]:
        """Dense joint covariance Σ_x = Λ_x⁻¹ of the flattened trajectory.

        Built column by column through `UDLDecomposition.solve`, so it
        costs O(T² D_x³) and is meant for small T (reference checks, dense
        downstream consumers).
        """
        # T D_x
        joint_size = self.horizon * self.state_dim
        identity = jnp.eye(joint_size, dtype=self.A.dtype)
        # Λ_x = U D Uᵀ
        udl = self.udl()
        # Σ_x = Λ_x⁻¹ I, one column at a time
        return jax.vmap(udl.solve, in_axes=1, out_axes=1)(identity)

    # ------------------------------------------------------------------
    # Precision views
    # ------------------------------------------------------------------

    def udl(self) -> UDLDecomposition:
        """The chain's precision, already factorised as Λ_x = U D Uᵀ.

        Returns:
            The `UDLDecomposition` with U_tᵀ = −F_t, D_t = Σ_η,t⁻¹ for
            t ≥ 1 and D_0 = Σ_x,0⁻¹.
        """
        return udl_from_ssm_params(self.A, self.Q, self.P0)

    @property
    def precision(self) -> BlockTriDiag:
        """Joint precision Λ_x = Σ_x⁻¹ as a `BlockTriDiag`.

        Raw Λ_x, not the −½ Λ_x natural-parameter convention of
        `ssm_to_naturals`.
        """
        return self.udl().as_block_tridiag()

    def to_precision_form(self) -> tuple[Float[Array, " Td"], BlockTriDiag]:
        """The chain as its flattened mean and joint precision ``(μ_x, Λ_x)``.

        This is the layout `spingp_posterior` consumes and returns: pass
        ``mean`` as its ``prior_mean`` (it assumes a zero-mean prior
        otherwise) and hand its output to `from_precision_form`. The
        information vector Λ_x μ_x, if needed, is ``precision.mv(mean)``.

        Returns:
            Tuple ``(mean, precision)``: the marginal means μ_x,t
            flattened state by state, and the joint precision Λ_x.
        """
        # μ_x = [μ_x,0; …; μ_x,T−1]
        mean = rearrange(self.mean, "step state -> (step state)")
        return mean, self.precision

    @classmethod
    def from_precision_form(
        cls,
        mean: Float[Array, " Td"] | Float[Array, "T d"],
        precision: BlockTriDiag,
    ) -> MarkovGaussian:
        """Recover the generative chain from its mean and precision.

            F_t = −U_tᵀ,   Σ_η,t+1 = D_t+1⁻¹,   c_t = μ_x,t+1 − F_t μ_x,t

        Round-trips `to_precision_form` exactly (to floating point).

        Algorithm:
            1. Factorise Λ_x = U D Uᵀ with one `udl_decomposition` pass.
            2. Read off F_t, Σ_η,t+1 and Σ_x,0 = D_0⁻¹ with
               `udl_to_ssm_params`.
            3. Recover the offsets c_t from the mean chain.

        Args:
            mean: Marginal means μ_x,t, flattened state by state or one row
                per state.
            precision: Symmetric positive-definite joint precision Λ_x.

        Returns:
            The equivalent `MarkovGaussian`.
        """
        num_blocks, block_size = precision._num_blocks, precision._block_size
        if mean.ndim == 2:
            means = mean
        else:
            # μ_x,t, one row per state
            means = rearrange(
                mean,
                "(step state) -> step state",
                step=num_blocks,
                state=block_size,
            )
        # Λ_x = U D Uᵀ
        udl = udl_decomposition(precision)
        # F_t = −U_tᵀ and the stacked covariances [Σ_x,0; Σ_η,1:T−1]
        transitions, covariances, _ = udl_to_ssm_params(udl)
        if num_blocks == 1:
            # One state, no transitions: einx cannot contract the empty axis.
            offsets = means[1:]
        else:
            # F_t μ_x,t
            propagated_means = einsum(
                transitions,
                means[:-1],
                "step next prev, step prev -> step next",
            )
            # c_t = μ_x,t+1 − F_t μ_x,t
            offsets = means[1:] - propagated_means
        return cls(transitions, covariances[1:], means[0], covariances[0], b=offsets)

    # ------------------------------------------------------------------
    # Density and sampling
    # ------------------------------------------------------------------

    def _chol_factors(self) -> tuple[Float[Array, "d d"], Float[Array, "Tm1 d d"]]:
        """Lower Cholesky factors L_x,0 of Σ_x,0 and L_η,t+1 of each Σ_η,t+1.

        Returns:
            Tuple ``(prior_factor, model_noise_factors)``.
        """
        # L_x,0 L_x,0ᵀ = Σ_x,0
        prior_factor = jnp.linalg.cholesky(self.P0)
        # L_η,t+1 L_η,t+1ᵀ = Σ_η,t+1
        model_noise_factors = jnp.linalg.cholesky(self.Q)
        return prior_factor, model_noise_factors

    def _log_prob_single(self, xs: Float[Array, "T d"]) -> Float[Array, ""]:
        """Log-density of one trajectory by the chain factorisation, O(T D_x³).

            log p(x_0:T−1) = log 𝒩(x₀; μ_x,0, Σ_x,0)
                + ∑_t log 𝒩(x_t+1; F_t x_t + c_t, Σ_η,t+1)

        Args:
            xs: Trajectory x_0:T−1, one row per state.

        Returns:
            The scalar log p(x_0:T−1).
        """
        prior_factor, model_noise_factors = self._chol_factors()
        # x₀ − μ_x,0
        prior_residual = xs[0] - self.mu0
        # log 𝒩(x₀; μ_x,0, Σ_x,0)
        prior_log_prob = _chol_log_prob(prior_factor, prior_residual)
        if self.A.shape[0] == 0:
            return prior_log_prob
        # F_t x_t
        propagated = einsum(self.A, xs[:-1], "step next prev, step prev -> step next")
        # x_t+1 − F_t x_t
        innovations = xs[1:] - propagated
        # r_t+1 = x_t+1 − F_t x_t − c_t
        residuals = innovations - self.b
        # log 𝒩(x_t+1; F_t x_t + c_t, Σ_η,t+1), one per transition
        transition_log_probs = jax.vmap(_chol_log_prob)(model_noise_factors, residuals)
        # ∑_t log 𝒩(x_t+1; F_t x_t + c_t, Σ_η,t+1), a full reduction
        transition_log_prob = jnp.sum(transition_log_probs)
        # log p(x_0:T−1)
        return prior_log_prob + transition_log_prob

    @validate_sample
    def log_prob(self, value: Float[Array, "*batch T d"]) -> Float[Array, "*batch"]:
        """Log-density log p(x_0:T−1) of each trajectory in ``value``.

        Args:
            value: Trajectories x_0:T−1, with leading batch axes.

        Returns:
            One log-density per trajectory.
        """
        leading_shape = value.shape[:-2]
        value_flat = rearrange(value, "... step state -> (...) step state")
        log_prob_flat = jax.vmap(self._log_prob_single)(value_flat)
        return _reshape_batch(log_prob_flat, leading_shape)

    def _sample_single(self, key: jax.Array) -> Float[Array, "T d"]:
        """One ancestral draw of the trajectory x_0:T−1, O(T D_x³).

            x₀ = μ_x,0 + L_x,0 z₀
            x_t+1 = F_t x_t + c_t + L_η,t+1 z_t+1

        Args:
            key: PRNG key.

        Returns:
            The trajectory x_0:T−1, one row per state.
        """
        prior_factor, model_noise_factors = self._chol_factors()
        num_states, state_size = self.horizon, self.state_dim
        prior_key, transition_key = jax.random.split(key)
        # Draw in the parameters' dtype, not JAX's default float.
        dtype = jnp.result_type(self.mu0, self.P0, self.Q)
        # z₀ ~ 𝒩(0, I)
        prior_noise = jax.random.normal(prior_key, (state_size,), dtype=dtype)
        # L_x,0 z₀
        prior_deviation = prior_factor @ prior_noise
        # x₀ = μ_x,0 + L_x,0 z₀
        initial_state = self.mu0 + prior_deviation
        # z_t+1 ~ 𝒩(0, I), one per transition
        transition_noise = jax.random.normal(
            transition_key, (num_states - 1, state_size), dtype=dtype
        )

        def _step(state, inputs):
            transition, offset, model_noise_factor, noise = inputs
            # F_t x_t
            propagated = transition @ state
            # F_t x_t + c_t
            mean_next = propagated + offset
            # L_η,t+1 z_t+1
            model_noise = model_noise_factor @ noise
            # x_t+1 = F_t x_t + c_t + L_η,t+1 z_t+1
            state_next = mean_next + model_noise
            return state_next, state_next

        _, states_rest = jax.lax.scan(
            _step,
            initial_state,

    def marginals(self) -> tuple[Float[Array, "T d"], Float[Array, "T d d"]]:
    """Marginal means μ_x,t and covariances Σ_x,t of every state.

        μ_x,t+1 = F_t μ_x,t + c_t
        Σ_x,t+1 = F_t Σ_x,t F_tᵀ + Σ_η,t+1

    Algorithm:
        1. Start from the prior moments μ_x,0 and Σ_x,0.
        2. Push the moments through each transition with a scan.
        3. Prepend the prior moments to the propagated ones.

    Returns:
        Tuple ``(means, covs)`` holding μ_x,t and Σ_x,t for
        t = 0, …, T − 1.
    """

    def _step(carry, inputs):
        mean, covariance = carry
        transition, offset, model_noise_covariance = inputs
        # F_t μ_x,t
        propagated_mean = transition @ mean
        # μ_x,t+1 = F_t μ_x,t + c_t
        mean_next = propagated_mean + offset
        # F_t Σ_x,t F_tᵀ, one three-operand contraction
        propagated_covariance = einsum(
            transition,
            covariance,
            transition,
            "next prev, prev col, other col -> next other",
        )
        # Σ_x,t+1 = F_t Σ_x,t F_tᵀ + Σ_η,t+1
        covariance_next = propagated_covariance + model_noise_covariance
        return (mean_next, covariance_next), (mean_next, covariance_next)

    _, (means_rest, covs_rest) = jax.lax.scan(
        _step, (self.mu0, self.P0), (self.A, self.b, self.Q)
    )
    # μ_x,0 as the first row
    prior_mean_row = rearrange(self.mu0, "state -> 1 state")
    # Σ_x,0 as the first slab
    prior_covariance_slab = rearrange(self.P0, "row col -> 1 row col")
    # μ_x,0:T−1, prior first
    means = jnp.concatenate([prior_mean_row, means_rest])
    # Σ_x,0:T−1, prior first
    covs = jnp.concatenate([prior_covariance_slab, covs_rest])
    return means, covs

        F_t = −U_tᵀ,   Σ_η,t+1 = D_t+1⁻¹,   c_t = μ_x,t+1 − F_t μ_x,t

        2. Read off F_t, Σ_η,t+1 and Σ_x,0 = D_0⁻¹ with

            per state.
        precision: Symmetric positive-definite joint precision Λ_x.

    Returns:
        The equivalent `MarkovGaussian`.

    if mean.ndim == 2:
        means = mean
    else:
        # μ_x,t, one row per state
        means = rearrange(
            mean,
            "(step state) -> step state",
            step=num_blocks,
            state=block_size,
        )
    # Λ_x = U D Uᵀ
    udl = udl_decomposition(precision)
    # F_t = −U_tᵀ and the stacked covariances [Σ_x,0; Σ_η,1:T−1]
    transitions, covariances, _ = udl_to_ssm_params(udl)

        offsets = means[1:]

        propagated_means = einsum(
            transitions,
            means[:-1],
            "step next prev, step prev -> step next",
        )
        # c_t = μ_x,t+1 − F_t μ_x,t
        offsets = means[1:] - propagated_means
    return cls(transitions, covariances[1:], means[0], covariances[0], b=offsets)
